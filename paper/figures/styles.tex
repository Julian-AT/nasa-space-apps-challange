% Shared TikZ styles for the paper diagrams (PRD 22.5). Input once from the main.tex preamble.
% Flat fills with thin dark outlines. The four fills differ in lightness as well as hue,
% so the workloads stay distinct in grayscale. No icons, gradients, shadows or images.
\usepackage{tikz}
\usetikzlibrary{positioning,fit,calc,arrows.meta}

\definecolor{fillPython}{HTML}{DCEBFA}  % offline Python, light blue
\definecolor{fillBundle}{HTML}{F4D77A}  % release bundle, saturated yellow, darkest
\definecolor{fillHosting}{HTML}{CDE8C6} % hosting, light green
\definecolor{fillBrowser}{HTML}{F2C3C3} % browser, pink
\definecolor{fillExternal}{HTML}{F2F2F2}% external product
\definecolor{lineDark}{HTML}{2B2B2B}

\tikzset{
  box/.style = {
    draw=lineDark, line width=0.5pt, rounded corners=1.5pt,
    font=\footnotesize, align=center, text=black,
    inner sep=3.5pt, minimum height=0.9cm, text width=#1
  },
  box/.default = 5.6cm,
  python/.style   = {box=#1, fill=fillPython},
  bundle/.style   = {box=#1, fill=fillBundle},
  hosting/.style  = {box=#1, fill=fillHosting},
  browser/.style  = {box=#1, fill=fillBrowser},
  external/.style = {box=#1, fill=fillExternal},
  python/.default = 5.6cm, bundle/.default = 5.6cm, hosting/.default = 5.6cm,
  browser/.default = 5.6cm, external/.default = 5.6cm,
  notdone/.style  = {dashed, dash pattern=on 3pt off 2pt},
  group/.style = {
    draw=lineDark, line width=0.4pt, rounded corners=2pt, inner sep=5pt
  },
  flow/.style = {-{Stealth[length=5pt,width=4pt]}, line width=0.6pt, draw=lineDark},
  flowdashed/.style = {flow, dashed, dash pattern=on 3pt off 2pt},
  flowlabel/.style = {font=\footnotesize, fill=white, inner sep=1.5pt, align=center}
}

% Plots (PRD 22.5): pgfplots reading generated number macros; colour-blind-safe Okabe-Ito hues.

% Layer diagram styles for the encoder figure: pastel fills after Figure 1 of Vaswani et al.
% (2017), fixed 8 pt and 7 pt type so the paper and the web copy match, thin dark outlines, a grey frame around the repeated block and small arrow heads.
\definecolor{tfOrange}{HTML}{FBDDB4}  % convolutions
\definecolor{tfYellow}{HTML}{F4F1BC}  % normalisation, activation, add
\definecolor{tfBlue}{HTML}{CBE5F3}    % pooling and MLP heads
\definecolor{tfPink}{HTML}{F6D3D7}    % input, stem, gradient reversal
\definecolor{tfGreen}{HTML}{D4EACB}   % losses
\definecolor{tfLilac}{HTML}{DEDAF0}   % embedding and total loss
\definecolor{tfFrame}{HTML}{EEEEEE}
\tikzset{
  layer/.style = {
    draw=lineDark, line width=0.45pt, rounded corners=3pt, font=\fontsize{8}{9.6}\selectfont,
    align=center, text=black, inner xsep=2pt, inner ysep=1.5pt,
    minimum height=0.5cm, text width=#1
  },
  layer/.default = 2.6cm,
  conv/.style = {layer=#1, fill=tfOrange}, conv/.default = 2.6cm,
  norm/.style = {layer=#1, fill=tfYellow}, norm/.default = 2.6cm,
  dense/.style = {layer=#1, fill=tfBlue},  dense/.default = 2.6cm,
  inlayer/.style = {layer=#1, fill=tfPink}, inlayer/.default = 2.6cm,
  lossbox/.style = {layer=#1, fill=tfGreen}, lossbox/.default = 2.6cm,
  embbox/.style = {layer=#1, fill=tfLilac}, embbox/.default = 2.6cm,
  addnode/.style = {
    circle, draw=lineDark, line width=0.45pt, fill=white, inner sep=0pt,
    minimum size=0.32cm, path picture={
      \draw[line width=0.45pt] ([xshift=-0.09cm]path picture bounding box.center)
        -- ([xshift=0.09cm]path picture bounding box.center);
      \draw[line width=0.45pt] ([yshift=-0.09cm]path picture bounding box.center)
        -- ([yshift=0.09cm]path picture bounding box.center);
    }
  },
  repframe/.style = {draw=lineDark, line width=0.45pt, fill=tfFrame, rounded corners=8pt},
  onnxframe/.style = {draw=lineDark, line width=0.4pt, rounded corners=6pt,
    dash pattern=on 2.5pt off 1.5pt},
  arr/.style = {-{Stealth[length=3.4pt,width=2.8pt]}, line width=0.5pt, draw=lineDark},
  wire/.style = {line width=0.5pt, draw=lineDark},
  junction/.style = {circle, fill=lineDark, inner sep=0pt, minimum size=2.4pt},
  note/.style = {font=\fontsize{7}{8.4}\selectfont, align=left, text=black, inner sep=1pt},
  wirelabel/.style = {font=\fontsize{7}{8.4}\selectfont, inner sep=1.5pt}
}
